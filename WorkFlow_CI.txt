name: NOME DO TESTE

on:
  push:
    branches:
      - main

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Baixar codigo
        uses: actions/checkout@v4

      - name: Configurar Pages
        uses: actions/configure-pages@v5

      - name: Publicar arquivos estaticos
        uses: actions/upload-pages-artifact@v3
        with:
          path: .          

      - name: Fazer deploy
        id: deployment
        uses: actions/deploy-pages@v4
